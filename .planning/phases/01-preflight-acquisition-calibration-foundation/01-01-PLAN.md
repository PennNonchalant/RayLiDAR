   set -euo pipefail
   ROOT="$(cd "$(dirname "${BASH_SOURCE[0]}")/.." && pwd)"
   if [[ ! -f "$ROOT/.env.container" ]]; then
     echo "ERROR: $ROOT/.env.container is missing." >&2
     echo "Create it from .env.container.example and set HLS_CONTAINER_RUN_CMD." >&2
     exit 78

   set -a; source "$ROOT/.env.container"; set +a
   : "${HLS_CONTAINER_RUN_CMD:?HLS_CONTAINER_RUN_CMD is not set}"
