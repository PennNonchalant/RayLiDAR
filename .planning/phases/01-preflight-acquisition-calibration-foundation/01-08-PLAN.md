   ```json
     "measured_at_utc": "YYYY-MM-DDTHH:MM:SSZ",
     "rectified_data_rect_only": true,
     "offset_unit": "original_1x_px",
     "frames": [
       {"frame": 0, "camera": "image_00", "features_measured": 5, "max_offset_px": 1.0, "notes": ""}
     ]
   }
   ```
