# Offline data pipeline: dataset acquisition/preflight, frame index, uncurl,
# masking, segment selection, GT cache, ray build/freeze, and G1/G2 checks.
# This target is intentionally separate from the runtime library; it only
# depends on core and the offline I/O dependencies.
file(GLOB_RECURSE RAYLIDAR_DATA_PROCESS_SOURCES CONFIGURE_DEPENDS
    "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp")

add_library(raylidar_data_process SHARED ${RAYLIDAR_DATA_PROCESS_SOURCES})
add_library(raylidar::data_process ALIAS raylidar_data_process)

target_compile_features(raylidar_data_process PUBLIC cxx_std_20)
set_target_properties(raylidar_data_process PROPERTIES
    CXX_VISIBILITY_PRESET hidden
    VISIBILITY_INLINES_HIDDEN ON)

# The export macro is generated by raylidar_core. Defining the core build macro
# here makes RAYLIDAR_EXPORT expand to dllexport while building this shared
# library on Windows (it is a no-op for ELF visibility on Linux).
target_compile_definitions(raylidar_data_process PRIVATE raylidar_core_EXPORTS)

target_include_directories(raylidar_data_process PUBLIC
    "${CMAKE_CURRENT_SOURCE_DIR}/include")

target_link_libraries(raylidar_data_process PUBLIC raylidar_core)

target_include_directories(raylidar_data_process PUBLIC "${Stb_INCLUDE_DIR}")

if(TARGET LibArchive::LibArchive)
    target_link_libraries(raylidar_data_process PUBLIC LibArchive::LibArchive)
else()
    target_link_libraries(raylidar_data_process PUBLIC ${LibArchive_LIBRARIES})
    target_include_directories(raylidar_data_process PUBLIC ${LibArchive_INCLUDE_DIRS})
endif()
